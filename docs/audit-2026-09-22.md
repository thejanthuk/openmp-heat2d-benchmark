## Appendix — the program behind the GPU/CPU ratio row

Both triads in one process, so both see the same machine state. STREAM accounting: **24 B/element**
(read `b`, read `c`, write `a`; the write-allocate read of `a` is not counted — multiply by 4/3 for bytes
actually moved). Best of 10. It refuses to report if the kernel ran on the host.

```sh
gcc -O3 -march=native -fopenmp -fno-stack-protector -foffload=nvptx-none \
    -foffload-options=nvptx-none="-O3 -fno-stack-protector -lm" triad_pair.c -lm -o triad_pair
for t in 1 4; do OMP_TARGET_OFFLOAD=MANDATORY OMP_NUM_THREADS=$t OMP_PROC_BIND=close ./triad_pair; done
```

```c
/* CPU vs GPU triad in one run, STREAM-counted (24 B/element), for the audit. */
#include <omp.h>
#include <stdio.h>
#include <stdlib.h>

#define N (1L << 25) /* 32 Mi doubles = 256 MB per array */
#define REPS 10

int main(void)
{
    double *a = malloc(N * sizeof *a), *b = malloc(N * sizeof *b), *c = malloc(N * sizeof *c);
    for (long i = 0; i < N; i++) { a[i] = 0.0; b[i] = 1.0; c[i] = 2.0; }
    const double s = 3.0, bytes = 24.0 * N;

    double cpu_best = 1e30;
    for (int r = 0; r < REPS; r++) {
        double t = omp_get_wtime();
        #pragma omp parallel for
        for (long i = 0; i < N; i++) a[i] = b[i] + s * c[i];
        t = omp_get_wtime() - t;
        if (t < cpu_best) cpu_best = t;
    }

    int on_host = -1;
    double gpu_best = 1e30;
    #pragma omp target data map(to: b[0:N], c[0:N]) map(tofrom: a[0:N])
    {
        #pragma omp target map(from: on_host)
        on_host = omp_is_initial_device();
        for (int r = 0; r < REPS; r++) {
            double t = omp_get_wtime();
            #pragma omp target teams distribute parallel for num_teams(10)
            for (long i = 0; i < N; i++) a[i] = b[i] + s * c[i];
            t = omp_get_wtime() - t;
            if (t < gpu_best) gpu_best = t;
        }
    }
    if (on_host != 0) { fprintf(stderr, "kernel ran on the HOST, refusing to report\n"); return 1; }

    double cpu = bytes / cpu_best / 1e9, gpu = bytes / gpu_best / 1e9;
    printf("threads=%d  CPU %.2f GB/s  GPU %.2f GB/s (%.1f%% of 112)  GPU/CPU %.2fx  check a[7]=%g\n",
           omp_get_max_threads(), cpu, gpu, 100 * gpu / 112, gpu / cpu, a[7]);
    free(a); free(b); free(c);
    return 0;
}
```
